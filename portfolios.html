    function activateAntiGravity() {
    cards.forEach((card, i) => {
    gsap.set(card, { transformOrigin: 'center center' });

    if (!activated) return;

    y: gsap.utils.random(-22, -4),
    rotate: gsap.utils.random(-2.2, 2.2),
    duration: gsap.utils.random(4.5, 7.5),
    ease: 'sine.inOut',
    onComplete: driftLoop
    });
    cardDrifts.set(card, tween);
    };
    const lift = gsap.fromTo(card, { y: 0 }, {
    y: gsap.utils.random(-18, -6),
    duration: 1.8, ease: 'power2.out', delay: i * 0.08,
    onComplete: driftLoop
    });
    cardDrifts.set(card, lift);
    });
    }

    line.setAttribute('x2', cx); line.setAttribute('y2', cy);
    line.style.opacity = 0;

    gsap.to(line.style, { opacity: 1, duration: 0.3, delay: i * 0.05 });

    opacity: 0, duration: 0.6, delay: 1.2 + i * 0.05,
    onComplete: () => line.remove()
    });
